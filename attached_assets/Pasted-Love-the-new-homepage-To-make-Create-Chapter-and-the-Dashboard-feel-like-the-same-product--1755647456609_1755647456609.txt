# 2) Reusable UI atoms (shadcn wrappers)

**Button.tsx**

```tsx
export function PrimaryButton(props: JSX.IntrinsicElements["button"]) {
  return (
    <button
      {...props}
      className={
        "inline-flex items-center justify-center rounded-full px-5 py-3 font-medium text-white " +
        "bg-gradient-to-r from-rr-accent to-rr-accentDark hover:opacity-90 focus:outline-none " +
        "focus:ring-2 focus:ring-rr-accent/40 " + (props.className ?? "")
      }
    />
  );
}

export function SecondaryButton(props: JSX.IntrinsicElements["button"]) {
  return (
    <button
      {...props}
      className={
        "inline-flex items-center justify-center rounded-xl px-4 py-2 text-sm " +
        "bg-white/5 border border-white/15 text-white hover:bg-white/10 " + (props.className ?? "")
      }
    />
  );
}
```

**Card.tsx**

```tsx
export function Card({children, className=""}:{children:React.ReactNode;className?:string}) {
  return (
    <div className={`rounded-2xl bg-rr-surface/95 border border-rr-border shadow-rrGlow ${className}`}>
      {children}
    </div>
  );
}
export const CardHeader = ({title, subtitle}:{title:string;subtitle?:string}) => (
  <div className="p-6 border-b border-rr-border">
    <h3 className="text-xl font-semibold">{title}</h3>
    {subtitle && <p className="text-rr-muted mt-1">{subtitle}</p>}
  </div>
);
export const CardBody = ({children}:{children:React.ReactNode}) => (
  <div className="p-6">{children}</div>
);
```

Use these everywhere (landing, learn-more, dashboard).

# 3) Create Chapter page (matches homepage)

**CreateChapter.tsx**

```tsx
import { PrimaryButton, SecondaryButton } from "@/components/ui/Button";
import { Card, CardHeader, CardBody } from "@/components/ui/Card";
import { useForm } from "react-hook-form";
import { z } from "zod"; import { zodResolver } from "@hookform/resolvers/zod";

const schema = z.object({
  chapterName: z.string().min(2),
  university: z.string().min(2),
  email: z.string().email(),
  logo: z.any().optional(),
});

export default function CreateChapter() {
  const { register, handleSubmit, formState:{errors,isSubmitting} } =
    useForm({ resolver: zodResolver(schema) });

  return (
    <main className="mx-auto max-w-6xl px-4 py-20">
      <header className="text-center mb-12">
        <h1 className="text-4xl md:text-5xl font-extrabold">Create your chapter</h1>
        <p className="text-rr-muted mt-3 max-w-2xl mx-auto">
          Set up RushRank for your house in minutes. Invite exec, import PNMs, and start your first event.
        </p>
      </header>

      <div className="grid md:grid-cols-3 gap-6">
        {/* Left: form */}
        <Card className="md:col-span-2">
          <CardHeader title="Chapter details" subtitle="You can change these later in Settings." />
          <CardBody>
            <form
              onSubmit={handleSubmit(() => {/* submit via TanStack mutation */})}
              className="grid gap-5"
            >
              <div>
                <label className="block text-sm text-rr-muted mb-1">Chapter name</label>
                <input {...register("chapterName")}
                  className="w-full rounded-xl bg-white/5 border border-rr-border px-4 py-3 focus:outline-none focus:ring-2 focus:ring-rr-accent/40" />
                {errors.chapterName && <p className="text-red-400 text-xs mt-1">Required</p>}
              </div>
              <div>
                <label className="block text-sm text-rr-muted mb-1">University</label>
                <input {...register("university")}
                  className="w-full rounded-xl bg-white/5 border border-rr-border px-4 py-3 focus:outline-none focus:ring-2 focus:ring-rr-accent/40" />
              </div>
              <div>
                <label className="block text-sm text-rr-muted mb-1">Admin email</label>
                <input {...register("email")}
                  className="w-full rounded-xl bg-white/5 border border-rr-border px-4 py-3" />
              </div>
              <div>
                <label className="block text-sm text-rr-muted mb-1">Logo (optional)</label>
                <input type="file" accept="image/*" {...register("logo")}
                  className="w-full rounded-xl bg-white/5 border border-rr-border px-4 py-2 file:mr-3 file:rounded-lg file:border-0 file:bg-white/10 file:text-white" />
              </div>

              <div className="flex gap-3 pt-2">
                <PrimaryButton disabled={isSubmitting}>Create Chapter</PrimaryButton>
                <SecondaryButton type="button">Invite Exec</SecondaryButton>
              </div>
            </form>
          </CardBody>
        </Card>

        {/* Right: helpful panel */}
        <Card>
          <CardHeader title="What you’ll set up" />
          <CardBody>
            <ul className="space-y-3 text-sm text-rr-muted">
              <li>• Roles & permissions (admin, exec, member)</li>
              <li>• Default rush stages & voting rules</li>
              <li>• Import PNMs via CSV or Google Sheet</li>
              <li>• First event with QR check-ins</li>
            </ul>
            <div className="mt-6 rounded-xl p-4 bg-rr-accentFaint">
              <div className="text-sm">Tip: you can switch to a demo chapter any time.</div>
            </div>
          </CardBody>
        </Card>
      </div>

      {/* Steps (reusing homepage style) */}
      <section className="mt-14 grid md:grid-cols-4 gap-4">
        {["Create chapter","Invite exec","Import PNMs","Start first event"].map((t,i)=>(
          <div key={t} className="rounded-2xl bg-white/5 border border-rr-border p-4">
            <div className="text-xs text-rr-muted">Step {i+1}</div>
            <div className="mt-1 font-semibold">{t}</div>
          </div>
        ))}
      </section>
    </main>
  );
}
```
